% -- shared macros ---------------------------------------------------------
% \code and \ep hold identifiers, paths and endpoints. Those are long
% unbreakable \texttt tokens that overflow the measure, so breakpoints are
% inserted after the characters they naturally break at. \ttbreak is applied
% by \lowercase-safe \scantokens-free means: a simple active-character-free
% approach, declaring the penalties \texttt already honours.
\newcommand{\code}[1]{\texttt{\small\hyphenchar\font=-1 #1}}
\newcommand{\ep}[1]{\texttt{\small\color{accent}\hyphenchar\font=-1 #1}}

% A boxed claim/finding, used for the things a marker should not miss.
% Collected into a save box first: \colorbox needs its argument complete, so
% it cannot be opened in the \begin and closed in the \end part.
\newsavebox{\kfbox}
\newenvironment{keyfinding}[1]
  {\begin{lrbox}{\kfbox}%
   \begin{minipage}{0.88\textwidth}%
   \textbf{\color{accent}#1}\par\vspace{0.7ex}\small\raggedright}
  {\end{minipage}%
   \end{lrbox}%
   \begin{center}\setlength{\fboxsep}{10pt}\colorbox{shade}{\usebox{\kfbox}}\end{center}}
